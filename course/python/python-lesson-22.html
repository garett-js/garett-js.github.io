Продвинутые методы работы с коллекциями в Python

Python предоставляет множество мощных инструментов для работы с коллекциями. В отличие от объектно-ориентированного подхода некоторых языков, Python сочетает функциональные возможности со встроенными функциями и специальным синтаксисом.

for цикл и итерирование элементов

В Python самый естественный способ обработки каждого элемента коллекции — использование for цикла.

Когда использовать: Когда нужно просто выполнить действие для каждого элемента, не изменяя саму коллекцию и не создавая новую.

Python
def main():
    languages = [
        'Python', 'JavaScript', 'Go', 'Rust', 'TypeScript'
    ]

    # Простой перебор
    for language in languages:
        print(f'Я изучаю {language}!')
    
    # С индексами через enumerate
    for index, language in enumerate(languages):
        print(f'{index + 1}. Я изучаю {language}!')

if __name__ == "__main__":
    main()

map() преобразование элементов

Функция map() применяет заданную функцию к каждому элементу итерируемого объекта и возвращает итератор с преобразованными элементами.

Python
def main():
    numbers = [1, 2, 3, 4, 5]
    languages = ['Python', 'JavaScript', 'Go', 'Rust', 'TypeScript']

    # Увеличиваем каждый элемент в 2 раза
    doubled = list(map(lambda x: x * 2, numbers))
    print(doubled)

    # Узнаём длину каждой строки
    lengths = list(map(len, languages))
    print(lengths)
    
    # Преобразование в верхний регистр
    uppercase = list(map(str.upper, languages))
    print(uppercase)

if __name__ == "__main__":
    main()

filter() фильтрация элементов

Функция filter() применяет функцию-предикат к каждому элементу и возвращает итератор, содержащий только те элементы, для которых функция вернула True.

Python
def main():
    students = [
        {'name': 'Артём', 'mark': 70},
        {'name': 'Анна', 'mark': 85},
        {'name': 'Наташа', 'mark': 65},
        {'name': 'Юля', 'mark': 90},
    ]

    # Все студенты которые набрали больше 70 баллов
    good_students = list(filter(lambda student: student['mark'] >= 70, students))
    print("Студенты с баллами >= 70:")
    print(good_students)

    # Все студенты имена которых начинаются с буквы А
    a_names = list(filter(lambda student: student['name'].startswith('А'), students))
    print("\nСтуденты на букву А:")
    print(a_names)

if __name__ == "__main__":
    main()

next() и генераторы для поиска элементов

Для поиска первого элемента, соответствующего условию, Python предлагает next() с генераторными выражениями.

Python
def main():
    languages = ['Python', 'JavaScript', 'Go', 'Rust', 'TypeScript']
    
    # Найти первый элемент длина которого меньше 3
    try:
        short_language = next(lang for lang in languages if len(lang) < 3)
        print(f"Короткий язык: {short_language}")
    except StopIteration:
        print("Нет языка короче 3 символов")
    
    # С значением по умолчанию
    short_language_safe = next(
        (lang for lang in languages if len(lang) < 3), 
        'Нет такого элемента'
    )
    print(f"Результат с default: {short_language_safe}")
    
    # Поиск последнего элемента (через reversed)
    last_long = next(
        (lang for lang in reversed(languages) if len(lang) > 5),
        'Не найдено'
    )
    print(f"Последний длинный язык: {last_long}")

if __name__ == "__main__":
    main()

any() и all() проверка соответствия условиям

Python имеет встроенные функции для проверки условий:

  • any() возвращает True, если хотя бы один элемент соответствует условию
  • all() возвращает True, если все элементы соответствуют условию
Python
def main():
    numbers = [1, 2, 3, 4, 5]
    languages = ['Python', 'JavaScript', 'Go', 'Rust', 'TypeScript']

    # Есть ли четное число?
    has_even = any(number % 2 == 0 for number in numbers)
    print(f"Есть четные числа: {has_even}")

    # Все ли числа положительные?
    all_positive = all(number > 0 for number in numbers)
    print(f"Все числа положительные: {all_positive}")

    # Все языки программирования начинаются с заглавной буквы?
    all_capitalized = all(language[0].isupper() for language in languages)
    print(f"Все языки с заглавной буквы: {all_capitalized}")
    
    # Есть ли языки короче 3 символов?
    has_short = any(len(language) < 3 for language in languages)
    print(f"Есть короткие языки: {has_short}")

if __name__ == "__main__":
    main()

Результат выполнения:

def advanced_any_all():
    students = [
        {'name': 'Артём', 'marks': [70, 75, 80]},

functools.reduce() свертывание коллекции

В Python есть functools.reduce() для агрегации коллекции в одно значение.

Python
from functools import reduce

def main():
    numbers = [1, 2, 3, 4, 5]
    students = [
        {'name': 'Артём', 'mark': 70},
        {'name': 'Анна', 'mark': 85},
        {'name': 'Наташа', 'mark': 65},
        {'name': 'Юля', 'mark': 90},
    ]

    # Сумма всех значений через reduce
    total = reduce(lambda acc, x: acc + x, numbers)
    print(f"Сумма через reduce: {total}")
    
    # То же самое с начальным значением
    total_with_initial = reduce(lambda acc, x: acc + x, numbers, 0)
    print(f"Сумма с начальным значением: {total_with_initial}")

    # Получить строку всех имен студентов с баллами > 65
    good_students_names = [
        student['name'] 
        for student in students 
        if student['mark'] > 65
    ]
    
    if good_students_names:
        names_string = reduce(lambda acc, name: f"{acc}, {name}", good_students_names)
        print(f"Хорошие студенты: {names_string}")

if __name__ == "__main__":
    main()

def reduce_alternatives():
    numbers = [1, 2, 3, 4, 5]
    words = ['Python', 'is', 'awesome', 'language']
    
    # Сумма - лучше использовать sum()
    total_sum = sum(numbers)

    # Произведение
    from math import prod  # Python 3.8+
    try:
        product = prod(numbers)
        print(f"Произведение: {product}")
    except AttributeError:
        # Для старых версий Python
        product = reduce(lambda x, y: x * y, numbers, 1)

    # Соединение строк - лучше использовать join()
    sentence = ' '.join(words)

    # Максимум - лучше использовать max()

itertools для продвинутой работы с последовательностями

Модуль itertools предоставляет множество полезных функций для работы с итераторами.

Python
import itertools

def main():
    values = [-22, 1, 45, 68, 98, 0, 100, 2, 3, 42, 11]
    
    # takewhile - элементы до первого не удовлетворяющего условию
    before_zero = list(itertools.takewhile(lambda x: x != 0, values))
    print(f"До нуля: {before_zero}")
    
    # dropwhile - элементы после первого удовлетворяющего условию
    after_zero = list(itertools.dropwhile(lambda x: x != 0, values))
    print(f"От нуля и дальше: {after_zero}")
    
    # islice - взять n элементов (аналог take)
    first_four = list(itertools.islice(values, 4))
    print(f"Первые 4: {first_four}")
    
    # islice с start и stop
    middle_elements = list(itertools.islice(values, 2, 7))
    print(f"Элементы с 2 по 6: {middle_elements}")

if __name__ == "__main__":
    main()

import itertools
from collections import defaultdict

def advanced_itertools():
    # chain - объединение нескольких итераторов
    list1 = [1, 2, 3]
    list2 = [4, 5, 6]
    list3 = [7, 8, 9]
    
    combined = list(itertools.chain(list1, list2, list3))
    print(f"Объединенные списки: {combined}")
    
    # groupby - группировка соседних одинаковых элементов
    data = ['a', 'a', 'b', 'b', 'b', 'c', 'a', 'a']
    grouped = [(key, list(group)) for key, group in itertools.groupby(data)]
    print(f"Группировка: {grouped}")
    
    # accumulate - накопительные суммы
    numbers = [1, 2, 3, 4, 5]
    cumulative = list(itertools.accumulate(numbers))
    print(f"Накопительные суммы: {cumulative}")
    
    # compress - фильтрация по маске
    data = ['a', 'b', 'c', 'd', 'e']
    selectors = [1, 0, 1, 0, 1]
    filtered = list(itertools.compress(data, selectors))
    print(f"Фильтрация по маске: {filtered}")
    
    # combinations и permutations
    items = ['A', 'B', 'C']
    combinations = list(itertools.combinations(items, 2))
    permutations = list(itertools.permutations(items, 2))
    print(f"Сочетания: {combinations}")
    print(f"Размещения: {permutations}")

if __name__ == "__main__":
    advanced_itertools()

from collections import defaultdict
from itertools import groupby

def grouping_examples():
    students = [
        {'name': 'Артём', 'grade': 'A', 'subject': 'Math'},
        {'name': 'Анна', 'grade': 'B', 'subject': 'Math'},
        {'name': 'Наташа', 'grade': 'A', 'subject': 'Physics'},
        {'name': 'Юля', 'grade': 'A', 'subject': 'Math'},
        {'name': 'Петр', 'grade': 'B', 'subject': 'Physics'},
    ]
    
    # Группировка через defaultdict
    by_grade = defaultdict(list)
    for student in students:
        by_grade[student['grade']].append(student['name'])
    
    print("Группировка по оценкам:")
    for grade, names in by_grade.items():
        print(f"  {grade}: {', '.join(names)}")
    
    # Группировка через itertools.groupby (требует сортировки)
    sorted_by_subject = sorted(students, key=lambda s: s['subject'])
    by_subject = {
        subject: list(group) 
        for subject, group in groupby(sorted_by_subject, key=lambda s: s['subject'])
    }
    
    print("\nГруппировка по предметам:")
    for subject, group in by_subject.items():
        names = [s['name'] for s in group]
        print(f"  {subject}: {', '.join(names)}")

if __name__ == "__main__":
    grouping_examples()

def generator_examples():
    # Генератор квадратов
    def squares(n):
        for i in range(n):
            yield i ** 2
    
    # Генераторное выражение
    even_squares = (x for x in squares(10) if x % 2 == 0)
    
    print("Четные квадраты:")
    for square in even_squares:
        print(f"  {square}")
    
    # Бесконечный генератор
    def fibonacci():
        a, b = 0, 1
        while True:
            yield a
            a, b = b, a + b
    
    # Первые 10 чисел Фибоначчи
    fib_gen = fibonacci()
    first_ten_fib = [next(fib_gen) for _ in range(10)]
    print(f"Первые 10 чисел Фибоначчи: {first_ten_fib}")
    
    # Генератор с отложенным вычислением
    def process_large_data():
        for i in range(1000000):
            if i % 100000 == 0:
                yield f"Processed {i} items"
    
    # Обрабатываем только первые 3 результата
    processor = process_large_data()
    for i, result in enumerate(processor):
        if i >= 3:
            break
        print(result)

if __name__ == "__main__":
    generator_examples()

    # 2. Топ игроки (сортировка)
    top_players = sorted(players, key=lambda p: p['score'], reverse=True)[:3]
    print(f"\nТоп-3 игрока:")
    for i, player in enumerate(top_players, 1):
        print(f"  {i}. {player['name']}: {player['score']} очков")
    
    # 3. Группировка по классам
    players_by_class = defaultdict(list)
    for player in players:
        players_by_class[player['class']].append(player)
    
    print(f"\nИгроки по классам:")
    for class_name, class_players in players_by_class.items():
        avg_class_score = sum(p['score'] for p in class_players) / len(class_players)
        print(f"  {class_name}: {len(class_players)} игроков, средний счет: {avg_class_score:.0f}")
    
    # 4. Фильтрация высокоуровневых игроков
    high_level = [p for p in players if p['level'] >= 25]
    high_level_names = [p['name'] for p in high_level]
    print(f"\nВысокоуровневые игроки (25+): {', '.join(high_level_names)}")
    
    # 5. Проверки с any/all
    has_max_level = any(p['level'] >= 35 for p in players)
    all_scored = all(p['score'] > 1000 for p in players)
    print(f"\nЕсть игроки 35+ уровня: {has_max_level}")
    print(f"Все набрали больше 1000 очков: {all_scored}")
    
    # 6. Агрегация через reduce
    total_guild_power = reduce(
        lambda acc, player: {
            **acc,
            player['guild']: acc.get(player['guild'], 0) + player['score']
        },
        players,
        {}
    )
    print(f"\nСила гильдий:")
    for guild, power in sorted(total_guild_power.items(), key=lambda x: x[1], reverse=True):
        print(f"  {guild}: {power} очков")
    
    # 7. Поиск элементов
    best_warrior = next(
        (p for p in players if p['class'] == 'Warrior'),
        None
    )
    if best_warrior:
        print(f"\nПервый найденный воин: {best_warrior['name']}")
    
    # 8. Сложная фильтрация и преобразование
    elite_players = [
        {
            'name': p['name'],
            'power_rating': p['level'] * 10 + p['score'] // 100,
            'tier': 'Elite' if p['score'] > 2000 else 'Advanced' if p['score'] > 1500 else 'Beginner'
        }
        for p in players
        if p['level'] >= 25
    ]
    
    print(f"\nРейтинги элитных игроков:")
    for player in sorted(elite_players, key=lambda p: p['power_rating'], reverse=True):
        print(f"  {player['name']}: {player['power_rating']} ({player['tier']})")

if __name__ == "__main__":
    game_data_analysis()

import time
from functools import lru_cache

def performance_tips():
    # Большой список для тестов
    large_list = list(range(1000000))
    
    # 1. Генераторы vs списки для больших данных
    def time_operation(name, operation):
        start = time.time()
        result = operation()
        end = time.time()
        print(f"{name}: {end - start:.4f} секунд")
        return result
    
    print("Сравнение производительности:")
    
    # List comprehension
    time_operation(
        "List comprehension", 
        lambda: [x * 2 for x in large_list if x % 1000 == 0]
    )
    
    # Generator expression (ленивое вычисление)
    gen = time_operation(
        "Generator creation", 
        lambda: (x * 2 for x in large_list if x % 1000 == 0)
    )
    
    # Использование генератора
    time_operation(
        "Generator consumption", 
        lambda: list(gen)
    )
    
    # 2. Кэширование результатов
    @lru_cache(maxsize=128)
    def expensive_calculation(n):
        """Имитация дорогой операции"""
        time.sleep(0.01)
        return n ** 2
    
    print("\nКэширование:")
    # Первый вызов - медленно
    start = time.time()
    result1 = expensive_calculation(10)
    print(f"Первый вызов: {time.time() - start:.4f} сек")
    
    # Второй вызов - быстро (из кэша)
    start = time.time()
    result2 = expensive_calculation(10)
    print(f"Второй вызов: {time.time() - start:.4f} сек")

if __name__ == "__main__":
    performance_tips()

Заключение и рекомендации

Python предлагает множество способов работы с коллекциями:

  • ✅ List/Dict/Set Comprehensions - самый Pythonic способ для простых операций
  • ✅ Built-in функции (map(), filter(), any(), all()) - для функционального стиля
  • ✅ itertools - для сложных операций с итераторами
  • ✅ Генераторы - для экономии памяти с большими данными
  • ✅ functools.reduce() - только когда нет специализированной функции (sum(), max(), etc.)

Выбирайте инструмент по ситуации:

  • Для читаемости: comprehensions
  • Для производительности: генераторы и itertools
  • Для функционального стиля: map/filter/reduce
  • Для сложной логики: обычные циклы

Помните: "Простое лучше сложного" - основной принцип Python!
