Задача 3

Решение
# 1. Вывод нечетных чисел
print("Нечетные числа от 1 до 5000:")
for i in range(1, 5001, 2):
    print(i)

# 2. Сумма четных чисел
sum_even = 0
for i in range(2, 5001, 2):
    sum_even += i
print(f"\nСумма всех четных чисел от 1 до 5000: {sum_even}")

Задача 5

Решение
number = 123654987
reversed_number = 0

while number > 0:
    digit = number % 10
    reversed_number = (reversed_number * 10) + digit
    number //= 10

print(f"Было: 123654987")
print(f"Стало: {reversed_number}")

Задача 7

Решение
total_sum = 0
for number in range(100, 201):
    if number % 17 == 0:
        total_sum += number

print(f"Сумма чисел от 100 до 200, кратных 17: {total_sum}")
